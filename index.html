function finalizePaymentAssignment(choice) { if (choice === "payment") { // Αν επιλέξει μήνα πληρωμής const d = fromYMD(newPayment.date); exp.reportingMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`; } else if (choice === "due") { // Αν επιλέξει μήνα λήξης const d = fromYMD(exp.dueDate); exp.reportingMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`; } else if (choice === "next") { // παλιό σενάριο (τέλη του μήνα) const d = fromYMD(exp.dueDate); d.setMonth(d.getMonth() + 1); exp.reportingMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`; } else { // default → να μείνει στον ίδιο μήνα λήξης const d = fromYMD(exp.dueDate); exp.reportingMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`; } exp.payments.push(newPayment); // Επαναλαμβανόμενο έξοδο (μένει ίδιο όπως πριν) if (exp.repeatInterval > 0) { const baseDate = exp.repeatBase === "payment" ? fromYMD(newPayment.date) : fromYMD(exp.dueDate); const nextDueDate = addMonths(baseDate, exp.repeatInterval); const newExp = mkExpense( exp.title, exp.amount, exp.category, nextDueDate, false, exp.notes, exp.repeatInterval, exp.repeatBase ); state.expenses.unshift(newExp); } persist(); renderAll(); notify("Η πληρωμή καταγράφηκε με επιτυχία!", "success"); el("payDialog").close(); }
